# (0) Se placer a la racine du depot (adapter le chemin si besoin).
cd ~/Distributed-Computing-Project


# --- TERMINAL A : deploiement + execution distribuee (8 workers, 8 splits) ---
# Lit la liste de machines deja verifiees, demarre le master + 8 workers, puis
# STREAME le log du master (MAP -> REDUCE) jusqu'a la fin (~45-60 s).
# >>> NOTER le master affiche : "Master etabli sur tp-XXXX:54321".
bash src/benchmarks/demo_verified_run.sh -w 8 -r 8 -s 8 -j wordcount


# --- TERMINAL B : TOLERANCE AUX PANNES (pendant que A tourne, phase MAP) ------
# Tue 1 worker en plein vol (UNIQUEMENT nos process, jamais ceux des autres).
# Dans le terminal A on voit alors : "Worker disconnected" -> MAP reassigne
# -> "job finished". Le job se termine quand meme, resultat correct.
bash src/deploy/fault_tolerance_demo.sh -n 1


# --- TERMINAL A : montrer le resultat (top mots par reducer, instantane) ------
# Remplacer tp-XXXX par le master note plus haut. Chaque part-*.txt est deja
# trie par frequence decroissante => head montre les mots les plus frequents.
ssh tp-XXXX "head -n 3 ~/slr207-group1-commoncrawl-$USER/output/part-*.txt"


# --- NETTOYAGE (garde les splits sur le NFS pour pouvoir rejouer la demo) ------
bash src/deploy/kill_commoncrawl.sh
